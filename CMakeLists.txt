cmake_minimum_required(VERSION 3.20)
project(HerramientasCpp LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(diseno src/main.cpp)
add_executable(pruebas tests/test.cpp)
target_include_directories(pruebas PRIVATE src)
if(MSVC)
  target_compile_options(diseno PRIVATE /utf-8)
  target_compile_options(pruebas PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME funciones_diseno COMMAND pruebas)
